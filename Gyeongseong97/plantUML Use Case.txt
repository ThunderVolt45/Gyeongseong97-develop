@startuml
left to right direction
actor "User" as user

rectangle "Gyeongseong 97" {

    package "Title Screen" {
        usecase "Start Game" as UC_Start
        usecase "View Credits" as UC_Credits
        usecase "Exit Application" as UC_Exit
    }

    package "Gameplay" {
        usecase "Control Player" as UC_Control
        usecase "Move (Arrow Keys)" as UC_Move
        usecase "Attack (Spacebar)" as UC_Attack
        usecase "Use Skill/Grenade" as UC_Skill
        usecase "Acquire Item" as UC_Item
        
        usecase "Defeat Enemies" as UC_Combat
        usecase "Fight Boss" as UC_BossBattle
        
        usecase "Game Over" as UC_GameOver
        usecase "Game Clear" as UC_GameClear
    }

    package "Game System" {
        usecase "Spawn Enemy Waves" as UC_Spawn
        usecase "Update Score" as UC_Score
        usecase "Play Sound/BGM" as UC_Sound
        usecase "Render TUI" as UC_Render
    }
}

' User interactions
user --> UC_Start
user --> UC_Credits
user --> UC_Exit
user --> UC_Control

' Relationships
UC_Start ..> UC_Control : <<include>>
UC_Control ..> UC_Move : <<include>>
UC_Control ..> UC_Attack : <<include>>
UC_Control ..> UC_Skill : <<extend>>

UC_Attack --> UC_Combat
UC_Skill --> UC_Combat

UC_Combat ..> UC_Score : <<include>>
UC_Combat ..> UC_Item : <<extend>> (Drop Item)
UC_Item ..> UC_Attack : <<extend>> (Change Weapon)

UC_Combat --> UC_BossBattle : Progress
UC_BossBattle --> UC_GameClear : On Success
UC_Control --> UC_GameOver : On Player Death

' System background processes
UC_Start --> UC_Spawn
UC_Start --> UC_Sound
UC_Start --> UC_Render

@enduml
