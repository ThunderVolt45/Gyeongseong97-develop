package "Core" {
    class GameObject {
        - bool active
        + float x
        + float y
        + Sprite sprite
        + GameObject()
        + virtual ~GameObject()
        + virtual void Update()
        + virtual void OnDestroy(std::shared_ptr<GameObject> self)
        + bool IsColliding(const GameObject& other)
        + virtual void OnCollision(GameObject& other)
        + virtual void Destroy()
    }

    class Player {
        + std::shared_ptr<Weapon> currentWeapon
        + std::shared_ptr<Weapon> defaultWeapon
        + std::shared_ptr<Weapon> grenadeWeapon
        + float maxHealth
        + float health
        + void Update() override
        + void OnCollision(GameObject& other) override
        + void Shoot()
        + void Grenade()
        + void EquipWeapon(std::shared_ptr<Weapon> newWeapon)
    }

    class Enemy {
        + float health
        + float speed
        + int killScore
        + void Update() override
        + void OnCollision(GameObject& other) override
        + virtual void TakeDamage(float damage)
    }

    class Bullet {
        + int damage
        + float speedX
        + float speedY
        + bool isPlayer
        + void Update() override
    }

    class Explosion {
        + int damage
        + void Update() override
        + void OnCollision(GameObject& other) override
    }

    class WeaponItem {
        + float speed
        + WeaponType weaponType
        + void Update() override
        + void OnCollision(GameObject& other) override
    }
}

package "Managers" {
    class GameManager <<Singleton>> {
        - StageManager stageManager
        - std::list<std::shared_ptr<GameObject>> gameObjects
        + {static} GameManager& GetInstance()
        + Player player
        + void Update()
        + void CreateGameObject(std::shared_ptr<GameObject> gameObject)
        + void DestroyGameObject(GameObject* gameObject)
    }

    class StageManager {
        - std::vector<Wave> waves
        + void Initialize()
        + void Update()
        + void Reset()
    }

    class AudioManager <<Singleton>> {
        + {static} AudioManager& GetInstance()
        + void PlayAudio(std::wstring audioPath, float volume, bool loop)
    }

    class InputManager <<Singleton>> {
        + {static} InputManager& GetInstance()
        + void Update()
        + bool IsKeyDown(int key_code)
        + bool IsKeyPressed(int key_code)
    }

    class RenderSystem {
        + {static} ftxui::Element Render()
        + {static} void ShowErrorMessage(const std::string& message)
    }

    class CollisionManager {
        + {static} void ProcessCollisions(...)
    }

    class ImageLoader {
        + {static} Sprite CreateSpriteFromImage(...)
    }

    class BulletPool <<Singleton>> {
        + {static} BulletPool& GetInstance()
        + std::shared_ptr<Bullet> GetBullet(...)
        + void ReturnBullet(std::shared_ptr<Bullet> bullet)
    }

    class ExplosionPool <<Singleton>> {
        + {static} ExplosionPool& GetInstance()
        + std::shared_ptr<Explosion> GetExplosion(...)
        + void ReturnExplosion(std::shared_ptr<Explosion> explosion)
    }
    
    class EnemyFactory {
        + {static} void CreateEnemy(EnemyInfo enemy)
    }

    class ItemFactory {
        + {static} void CreateItem(ItemInfo item)
    }
}

package "Enemies" {
    class Vanguard {
        + void Update() override
    }

    class Carmikaze {
        + void Update() override
    }

    class Narration {
        + BossState* state
        + void Update() override
        + void ChangePattern()
    }
}

package "Weapons" {
    abstract class Weapon {
        + WeaponType type
        + int damage
        + int maxBullet
        + int remainBullet
        + virtual void Shoot(Player* owner)
    }

    class WeaponDefault {
        + void Shoot(Player* owner) override
    }

    class WeaponHMG {
        + void Shoot(Player* owner) override
    }

    class WeaponShotgun {
        + void Shoot(Player* owner) override
    }

    class WeaponGrenade {
        + void Shoot(Player* owner) override
    }
}

package "BossStates" {
    abstract class BossState {
        + virtual void Update(Narration& boss)
    }

    class BossStateAppeared
    class BossStateIdle
    class BossStateAttackShot
    class BossStateAttackDive
    class BossStateAttackBomb
    class BossStateSpawnVanguard
    class BossStateSpawnCarmikaze
    class BossStateDeath
}

Weapon <|-- WeaponGrenade

BossState <|-- BossStateIdle
BossState <|-- BossStateAttackShot
BossState <|-- BossStateAttackDive
BossState <|-- BossStateAttackBomb
BossState <|-- BossStateSpawnVanguard
BossState <|-- BossStateSpawnCarmikaze
BossState <|-- BossStateDeath

' Association & Composition
GameManager o-- GameObject
GameManager *-- StageManager
GameManager *-- Player

Player o-- Weapon
Narration o-- BossState

BulletPool ..> Bullet : manages
ExplosionPool ..> Explosion : manages

EnemyFactory ..> Enemy : creates
ItemFactory ..> WeaponItem : creates
