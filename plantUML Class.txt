package "Weapons" {

        + Default
        + HMG
        + Shotgun
        + Grenade
        + Rocket

    abstract class Weapon {
        + WeaponType type;
        + int damage;
        + int maxBullet;
        + int remainBullet;
        + bool isInfinite;
        + int cooldownTick;
        + damage : int
        + maxBullet : int
        + {abstract} virtual void Shoot(Player* owner)
        + void AddAmmo(int bullet)
        + bool IsEmpty() const
    }

    class WeaponDefault {

    class WeaponHMG {
